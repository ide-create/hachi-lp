rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn() &&
        get(/databases/$(database)/documents/staff/$(request.auth.uid)).data.role == 'admin';
    }
    match /staff/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }
    match /punches/{id} {
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }
    match /reports/{id} {
      allow create, update: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow delete: if isAdmin();
    }
    match /messages/{id} {
      allow create: if signedIn() &&
        (request.resource.data.uid == request.auth.uid || isAdmin());
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }
  }
}
